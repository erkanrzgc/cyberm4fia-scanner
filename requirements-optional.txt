# ── MITM proxy interceptor (--proxy-listen) ────────────────────────────
# bcrypt is pinned <4.1 because newer bcrypt + the current passlib release
# raise "ValueError: password cannot be longer than 72 bytes" at mitmproxy
# import time. Remove the pin once passlib ships the truncation patch.
mitmproxy>=10.0
passlib>=1.7.4
bcrypt>=3.2.0,<4.1

# Used by utils/qishing.py + modules/guaranteed_checks.py to scan images
# embedded on the target page for malicious QR codes.
Pillow>=10.0
pyzbar>=0.1.9

# ── SSH / FTP brute-force (modules/brute_force.py) ─────────────────────
# Required when --spray runs against an SSH or FTP service.
paramiko>=3.4.0
